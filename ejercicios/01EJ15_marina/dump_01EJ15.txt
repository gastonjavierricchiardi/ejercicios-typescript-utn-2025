     9

    11	        this._navios.push(navio);
    12	    }
    13	
    14	    public cantidad(): number {
    15	        return this._navios.length;
    16	    }
    17	
    18	    public buscarPorNombre(nombre: string): Navio | undefined {
    19	        return this._navios.find(n => n.getNombre() === nombre);
    20	    }
    21	
    22	    // Muestra SOLO lo común del Navio (sin meterse con tipos concretos)

    24	        console.log(`--- Flota (${this.cantidad()} navíos) ---`);
    25	        for (const navio of this._navios) {

    27	                `Nombre: ${navio.getNombre()} | ` +
    28	                `Flotabilidad: ${navio.getFlotabilidad()} | ` +
    29	                `Solidez: ${navio.getSolidez()} | ` +
    30	                `Estabilidad: ${navio.getEstabilidad()} | ` +
    31	                `Vel. Crucero: ${navio.getVelocidadCrucero()}`
